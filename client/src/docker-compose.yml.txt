version: '3.8'

services:
  frontend:
    build: ./client
    ports:
      - "3000:3000"
    depends_on:
      - backend
    environment:
      - REACT_APP_API_URL=http://backend:5000/api

  backend:
    build: ./server
    ports:
      - "5000:5000"
    environment:
      - DB_USER=${DB_USER}
      - DB_HOST=db
      - DB_NAME=${DB_NAME}
      - DB_PASSWORD=${DB_PASSWORD}
      - DB_PORT=${DB_PORT}
      - JWT_SECRET=${JWT_SECRET}
      - KRAKEN_API_KEY=${KRAKEN_API_KEY}
      - KRAKEN_PRIVATE_KEY=${KRAKEN_PRIVATE_KEY}
      - INFURA_PROJECT_ID=${INFURA_PROJECT_ID}
      - METAMASK_PRIVATE_KEY=${METAMASK_PRIVATE_KEY}
    depends_on:
      - db

  db:
    image: postgres:13
    environment:
      - POSTGRES_USER=${DB_USER}
      - POSTGRES_PASSWORD=${DB_PASSWORD}
      - POSTGRES_DB=${DB_NAME}
    volumes:
      - postgres_data:/var/lib/postgresql/data
